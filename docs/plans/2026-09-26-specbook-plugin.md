**Tech Stack:** Browser ES modules, Studio SDK (`/studio-plugin.js`: `studio.project/onProjectChange/me/records/xlsx/docx/download/notify/confirm/prompt`), studio-kit CSS, Node 26 `node:test` for the pure modules, headless Chrome for the DOCX smoke test.

```js
const SHEET = '預算詳細表';
const HEADER = '編碼(備註)';
const ITEM_CODE = /^[0-9A-Z]{5,}$/;

const compact = (value) => String(value ?? '').replace(/\s+/g, '');

// exceljs cell values: string, number, {richText:[{text}]}, or {formula, result}.
function cellText(value) {
  if (value && typeof value === 'object') {
    if (Array.isArray(value.richText)) return value.richText.map((part) => part.text).join('');
    if ('result' in value) return cellText(value.result);
    return '';
  }
  return String(value ?? '');
}

// Item codes from the PCCES detail sheet's 編碼(備註) column. Flags after a comma (",#,*") are dropped;
// numeric cells are reported instead of guessed because Excel already removed their leading zeros.
export function extractItemCodes(sheets) {
  const rows = sheets[SHEET];
  if (!rows) return { codes: [], warnings: [], error: `找不到「${SHEET}」工作表。` };
  let headerRow = -1;
  let column = -1;
  for (let r = 0; r < rows.length && column < 0; r += 1) {
    column = (rows[r] ?? []).findIndex((cell) => compact(cellText(cell)) === HEADER);
    headerRow = r;
  }
  if (column < 0) return { codes: [], warnings: [], error: `「${SHEET}」沒有「${HEADER}」欄。` };

  const codes = [];
  const warnings = [];
  for (let r = headerRow + 1; r < rows.length; r += 1) {
    const cell = (rows[r] ?? [])[column];
    if (typeof cell === 'number') {
      warnings.push(`第 ${r + 1} 列的編碼是數字格式（${cell}），開頭的 0 可能已遺失，已略過，請改用文字格式或手動加章。`);
      continue;
    }
    const code = cellText(cell).split(',')[0].trim().toUpperCase();
    if (ITEM_CODE.test(code) && !codes.includes(code)) codes.push(code);
  }
  return { codes, warnings, error: null };
}
```

// Minimal in-memory stand-in for studio.records (SDK notes §2): versions, CAS, VERSION_CONFLICT with err.current.
export class FakeStudio {
  constructor() {
    this.rows = new Map();
    this.calls = [];
    this.records = {
      getByKey: async (collection, key, opts) => {
        this.calls.push(['getByKey', collection, key, opts]);
        return structuredClone(this.rows.get(`${collection}/${key}`) ?? null);
      },
      put: async (collection, key, { scope, project_handle, body, version }) => {
        this.calls.push(['put', collection, key, { scope, project_handle, version }]);
        const id = `${collection}/${key}`;
        const current = this.rows.get(id);
        if (current && version !== undefined && version !== current.version) {
          throw Object.assign(new Error('VERSION_CONFLICT'), { current: structuredClone(current) });
        }
        const next = { record_id: id, key, body, version: (current?.version ?? 0) + 1, state: 'draft', updated_by_name: 'tester' };
        this.rows.set(id, next);
        return structuredClone(next);
      },
      publish: async (collection, recordId, { version }) => {
        this.calls.push(['publish', collection, recordId, version]);
        this.rows.get(recordId).state = 'published';
      },
      reopen: async (collection, recordId, { version, reason }) => {
        this.calls.push(['reopen', collection, recordId, version, reason]);
        this.rows.get(recordId).state = 'reopened';
      },
    };
  }
}

```js
export const PROJECT = 'specbook-project';
export const CHAPTER = 'specbook-chapter';
const PROJECT_KEY = 'main';

// One project record (key "main") and one record per chapter (key = chapter code), all scope "project"
// (spec §6.2). A VERSION_CONFLICT is returned to the caller, never retried (SDK notes §2, pattern b).
export function createStore(studio, projectHandle) {
  const scoped = { scope: 'project', project_handle: projectHandle };

  async function put(collection, key, body, version) {
    try {
      const record = await studio.records.put(collection, key, { ...scoped, body, version });
      return { ok: true, version: record.version, recordId: record.record_id };
    } catch (error) {
      if (error.message !== 'VERSION_CONFLICT') throw error;
      const { body: current, version: currentVersion, updated_by_name: by } = error.current;
      return { ok: false, conflict: { body: current, version: currentVersion, by } };
    }
  }

  async function get(collection, key) {
    const record = await studio.records.getByKey(collection, key, scoped);
    return record && { body: record.body, version: record.version, recordId: record.record_id, state: record.state };
  }

  return {
    loadProject: () => get(PROJECT, PROJECT_KEY),
    saveProject: (body, version) => put(PROJECT, PROJECT_KEY, body, version),
    async loadChapter(code) {
      const record = await get(CHAPTER, code);
      return record && { decisions: record.body.decisions, version: record.version, recordId: record.recordId, state: record.state };
    },
    async saveChapter(code, decisions, version) {
      const result = await put(CHAPTER, code, { decisions }, version);
      if (result.ok) return result;
      return { ok: false, conflict: { decisions: result.conflict.body.decisions, version: result.conflict.version, by: result.conflict.by } };
    },
    async publishAll(entries) {
      for (const { collection, recordId, version } of entries) await studio.records.publish(collection, recordId, { version });
    },
    async reopenAll(entries, reason) {
      for (const { collection, recordId, version } of entries) await studio.records.reopen(collection, recordId, { version, reason });
    },
  };
}
```

// Engine DOCX model (spec §5.4) -> the data shape of templates/specbook.docx (build_template.py).
export function toTemplateData(model) {
  return {
    draftLabel: model.watermark ?? '',
    toc: model.toc.map(({ title }) => ({ title })),
    chapters: model.chapters.map((chapter) => ({
      title: chapter.title,
      paragraphs: chapter.paragraphs.map(({ number, text }) => ({ line: number ? `${number}\t${text}` : text })),
      end: chapter.end,
    })),
  };
}

import { chapterDefaults, decisionState, hasBlocking, lint, progress, toDocxModel } from './engine/index.js';
import { el, notice, showStep, state, stepHandlers } from './app.js';
import { toTemplateData } from './lib/docx-data.js';
import { CHAPTER, PROJECT } from './lib/store.js';

const $ = (id) => document.getElementById(id);
const DOCX_MIME = 'application/vnd.openxmlformats-officedocument.wordprocessingml.document';
const RULES = { unresolved: '尚未填寫', stale: '規範已改版，請重選', invalid_choice: '選項不適用，請重選' };
const chapters = new Map(); // code -> {decisions, version, recordId, state}
let current = null;

async function ensureChapter(code) {
  if (!chapters.has(code)) chapters.set(code, (await state.store.loadChapter(code)) ?? { decisions: {}, version: undefined });
  return chapters.get(code);
}

const decision = (chapter, id, choice) => ({
  slot: Number(id), chapter_version: chapter.version, choice, at: new Date().toISOString(), by: state.me.displayName,
});

async function save(code, decisions) {
  const record = chapters.get(code);
  const result = await state.store.saveChapter(code, decisions, record.version);
  if (result.ok) {
    chapters.set(code, { ...record, decisions, version: result.version, recordId: result.recordId });
    $('conflict').hidden = true;
  } else {
    showConflict(code, decisions, result.conflict);
  }
  renderProgress(code);
}

// SDK notes §2 pattern (b): show both versions; the user chooses, nothing is retried automatically.
function showConflict(code, mine, theirs) {
  const box = $('conflict');
  box.replaceChildren(
    el('p', {}, `本章剛被 ${theirs.by} 修改。請選擇要保留哪一份：`),
    el('div', { class: 'sb-row' },
      el('button', { class: 'kit-button kit-button--secondary', type: 'button', on: { click: () => {
        chapters.set(code, { ...chapters.get(code), decisions: theirs.decisions, version: theirs.version });
        box.hidden = true;
        openChapter(code);
      } } }, `改用 ${theirs.by} 的版本`),
      el('button', { class: 'kit-button kit-button--danger', type: 'button', on: { click: async () => {
        if (!(await studio.confirm('確定用你的內容覆蓋對方的修改？', { tone: 'danger', confirmLabel: '覆蓋' }))) return;
        chapters.set(code, { ...chapters.get(code), version: theirs.version });
        await save(code, mine);
      } } }, '用我的版本覆蓋')),
  );
  box.hidden = false;
}

function renderProgress(code) {
  const { filled, total } = progress(state.pack, code, chapters.get(code).decisions);
  $('progress').textContent = `已填 ${filled}/${total}`;
}

function sentence(chapter, paragraph, slotId) {
  const p = el('p', { class: 'sb-sentence' }, paragraph.number ? `${paragraph.number}　` : '');
  for (const part of paragraph.text) {
    if (typeof part === 'string') p.append(part);
    else if (String(part.slot) === slotId) p.append(el('mark', {}, chapter.slots[slotId].raw));
    else p.append(chapter.slots[String(part.slot)].raw);
  }
  return p;
}

function controls(code, chapter, slotId, decisions) {
  const slot = chapter.slots[slotId];
  const record = decisions[slotId];
  const status = decisionState(chapter, slotId, record);
  const choose = (choice) => save(code, { ...chapters.get(code).decisions, [slotId]: decision(chapter, slotId, choice) });
  const wrap = el('div', { class: 'sb-group' });
  if (status === 'stale' || status === 'invalid') wrap.append(el('p', { class: 'kit-field-error' }, status === 'stale' ? '規範已改版，請重選。' : '原本的選擇不適用，請重選。'));
  const picked = status === 'ok' ? record.choice : null;

  if (slot.kind === 'malformed') {
    const area = el('textarea', { 'aria-label': '改寫整段', disabled: state.readOnly ? '' : null });
    area.value = picked?.rewrite ?? slot.raw;
    area.addEventListener('change', () => area.value.trim() && choose({ rewrite: area.value }));
    wrap.append(el('p', { class: 'sb-meta' }, '此段括號不成對，請直接改寫整段。'), area);
    return wrap;
  }
  const options = slot.options.map((text, index) => [String(index), text]);
  if (slot.kind === 'optional') options.push(['omit', '刪除此句']);
  if (options.length) {
    const group = el('fieldset', { class: 'kit-segmented' }, el('legend', { class: 'kit-sr-only' }, '選項'));
    for (const [value, text] of options) {
      const input = el('input', { type: 'radio', name: `slot-${code}-${slotId}`, value, disabled: state.readOnly ? '' : null });
      input.checked = picked ? (value === 'omit' ? picked.omit === true : picked.option === Number(value)) : false;
      input.addEventListener('change', () => choose(value === 'omit' ? { omit: true } : { option: Number(value) }));
      group.append(el('label', {}, input, text));
    }
    wrap.append(group);
  }
  if (['default_blank', 'blank', 'choice_blank'].includes(slot.kind)) {
    const id = `text-${code}-${slotId}`;
    const input = el('input', { id, type: 'text', disabled: state.readOnly ? '' : null });
    input.value = picked?.text ?? '';
    input.addEventListener('change', () => input.value.trim() && choose({ text: input.value.trim() }));
    wrap.append(el('div', { class: 'kit-field' }, el('label', { for: id }, '自行填寫'), input));
  }
  return wrap;
}

async function openChapter(code) {
  current = code;
  const record = await ensureChapter(code);
  const chapter = state.pack.chapters[code];
  const pick = $('chapter-pick');
  pick.replaceChildren(...state.chosen.map((c) => el('option', { value: c }, `${c} ${state.pack.chapters[c].name}`)));
  pick.value = code;
  const blocks = [];
  for (const paragraph of chapter.paragraphs) {
    for (const part of paragraph.text) {
      if (typeof part === 'string') continue;
      const slotId = String(part.slot);
      blocks.push(el('div', { class: 'sb-slot' }, sentence(chapter, paragraph, slotId), controls(code, chapter, slotId, record.decisions)));
    }
  }
  $('slots').replaceChildren(...(blocks.length ? blocks : [el('p', { class: 'sb-meta' }, '本章沒有需要填寫的括號。')]));
  $('defaults').disabled = state.readOnly;
  renderProgress(code);
  showStep(3);
}

async function applyDefaults() {
  const record = chapters.get(current);
  const next = chapterDefaults(state.pack, current, record.decisions, { at: new Date().toISOString(), by: state.me.displayName });
  await save(current, next);
  await openChapter(current);
}

async function allDecisions() {
  const result = {};
  for (const code of state.chosen) result[code] = (await ensureChapter(code)).decisions;
  return result;
}

async function renderLint() {
  const decisions = await allDecisions();
  const issues = lint(state.pack, state.chosen, decisions);
  const blocking = hasBlocking(issues);
  const byChapter = new Map();
  for (const issue of issues) {
    if (!byChapter.has(issue.chapter)) byChapter.set(issue.chapter, []);
    byChapter.get(issue.chapter).push(issue);
  }
  const items = [...byChapter].map(([code, list]) => {
    const counts = {};
    for (const i of list.filter((x) => x.blocking)) counts[i.rule] = (counts[i.rule] ?? 0) + 1;
    const notes = list.filter((x) => !x.blocking).map((x) => (x.rule === 'chapter_defaults' ? `${x.count} 格以「全部採用預設」帶入` : `版次疑義：${x.note}`));
    const summary = [...Object.entries(counts).map(([rule, n]) => `${RULES[rule]} ${n} 格`), ...notes].join('；');
    return el('div', { class: 'kit-list-item sb-row' }, el('span', {}, `${code} ${state.pack.chapters[code].name}：${summary}`),
      el('button', { class: 'kit-button kit-button--link', type: 'button', on: { click: () => openChapter(code) } }, '前往'));
  });
  $('lint').replaceChildren(
    el('div', { class: 'kit-callout', 'data-status': blocking ? 'warning' : 'success' }, blocking ? '還有未完成的格子，只能下載草稿。' : '全部格子都已處理，可以下載正式版。'),
    el('div', { class: 'kit-list' }, ...items),
  );
  $('download-final').disabled = blocking;
  $('publish').disabled = blocking || state.readOnly;
  $('publish').hidden = state.readOnly;
  $('reopen').hidden = !state.readOnly;
  showStep(4);
}

async function download(draft) {
  const model = toDocxModel(state.pack, state.chosen, await allDecisions(), { draft });
  const template = await (await fetch('./templates/specbook.docx')).blob();
  const blob = await studio.docx.fromTemplate(template, toTemplateData(model));
  const name = `${state.project?.name ?? '施工規範'}${draft ? '-草稿' : ''}.docx`;
  await studio.download(name, blob, DOCX_MIME);
}

async function entries() {
  const list = [{ collection: PROJECT, recordId: state.projectRecord.recordId, version: state.projectRecord.version }];
  for (const code of state.chosen) {
    const record = await ensureChapter(code);
    if (record.recordId) list.push({ collection: CHAPTER, recordId: record.recordId, version: record.version });
  }
  return list;
}

async function publish() {
  if (!(await studio.confirm('定稿後所有人都不能再修改，確定？', { confirmLabel: '定稿' }))) return;
  await state.store.publishAll(await entries());
  state.projectRecord = await state.store.loadProject(); // publish bumps versions server-side
  state.readOnly = true;
  chapters.clear();
  studio.notify('已定稿');
  await renderLint();
}

async function reopen() {
  const reason = await studio.prompt('解除定稿的原因（1–200 字）');
  if (!reason) return;
  await state.store.reopenAll(await entries(), reason.slice(0, 200));
  state.projectRecord = await state.store.loadProject(); // reopen bumps versions server-side
  state.readOnly = false;
  chapters.clear();
  await renderLint();
}
